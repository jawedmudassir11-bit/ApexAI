Go to Settings > Safari > Advanced > Feature Flags \n2. Turn ON 'WebGPU' \n3. Restart Safari."); } else { alert("Your browser does not support WebGPU, which is required for local AI. Please use Google Chrome, Edge, or Brave."); } return addMsgToDOM("WebGPU is blocked. Please enable it in settings or use Chrome.", "bot error", -1); } try { setStatus("CONNECTING...", 0); appendTelemetryLog(`Requesting GPU access for: ${currentModelId.split('-')[0]}`); // Safety timeout: If it stays on 0% for more than 15 seconds, tell the user. let loadingStarted = false; setTimeout(() => { if (!loadingStarted) { appendTelemetryLog("Warning: Download is hanging. Check your internet connection or clear the VRAM cache."); } }, 15000); engine = await webllm.CreateMLCEngine(currentModelId, { initProgressCallback: (p) => { loadingStarted = true; setStatus(`DOWNLOADING ${Math.round((p.progress||0)*100)}%`, Math.round((p.progress||0)*100)); if(p.text) appendTelemetryLog(p.text); } }); ready = true; setStatus("ONLINE", 100); input.disabled = false; input.focus(); appendTelemetryLog("Local compute matrix state: ONLINE."); } catch (error) { setStatus("CRASHED", 0); appendTelemetryLog(`Core error: ${error.message}`); // Catch specific out-of-memory errors common on iPhones/iPads if(error.message.includes("buffer") || error.message.includes("memory")) { alert("Your device ran out of memory trying to load this AI. \n\nClick 'Clear VRAM Cache' in the sidebar, and switch to the 'TinyLlama' model, which is much smaller."); } else { alert(`Initialization Failed: ${error.message}`); } addMsgToDOM("Engine failed to load. You may be out of memory. Try switching to TinyLlama.", "bot error", -1); } }
